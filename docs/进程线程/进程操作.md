# 进程操作

执行下面的代码，创建多个子进程并运行。

```python
from multiprocessing import Process


def func1():
    print('Hello World!')


def func2():
    print('Hello Python!')


def func3():
    print(233, 666)


if __name__ == '__main__':
    p1 = Process(target=func1)
    p2 = Process(target=func2)
    p3 = Process(target=func3)
    p1.start()
    p2.start()
    p3.start()
```

执行下面的代码，分别输出多个子进程的信息。

```python
from multiprocessing import Process
import os


def func1():
    print(os.getpid(), os.getppid())


def func2():
    print(os.getpid(), os.getppid())


def func3():
    print(os.getpid(), os.getppid())


if __name__ == '__main__':
    print(os.getpid(), os.getppid())
    p1 = Process(target=func1)
    p2 = Process(target=func2)
    p3 = Process(target=func3)
    p1.start()
    p2.start()
    p3.start()
```

在多进程环境中， CPU 会根据调度算法动态分配资源，这样会导致运行结果并不完全按照程序指令顺序执行。执行下面的代码，利用进程阻塞来确保进程执行与完成的先后顺序。

```python
from multiprocessing import Process


def func1():
    num = 0
    for i in range(1000000):
        num = i
    print('func1', num)


def func2():
    num = 0
    for i in range(1000000):
        num = i
    print('func2', num)


def func3():
    num = 0
    for i in range(1000000):
        num = i
    print('func3', num)


if __name__ == '__main__':
    p1 = Process(target=func1)
    p2 = Process(target=func2)
    p3 = Process(target=func3)
    p1.start()
    p1.join()
    p2.start()
    p2.join()
    p3.start()
```

在多进程环境中，进程间不共享全局变量。

```python
from multiprocessing import Process

num = 233


def func1():
    global num
    num += 1
    print(num)


def func2():
    global num
    num += 1
    print(num)


def func3():
    global num
    num += 1
    print(num)


if __name__ == '__main__':
    p1 = Process(target=func1)
    p2 = Process(target=func2)
    p3 = Process(target=func3)
    p1.start()
    p2.start()
    p3.start()
```

可以使用 Python 中提供的队列模块来实现进程间的变量共享。

```python
from multiprocessing import Process, Queue


def func1(queue):
    queue.put(1)
    queue.put(2)
    queue.put(3)


def func2(queue):
    print(queue.get())
    print(queue.get())


def func3(queue):
    print(queue.get())


if __name__ == '__main__':
    queue = Queue(3)
    p1 = Process(target=func1, args=(queue,))
    p2 = Process(target=func2, args=(queue,))
    p3 = Process(target=func3, args=(queue,))
    p1.start()
    p1.join()
    p2.start()
    p2.join()
    p3.start()
```

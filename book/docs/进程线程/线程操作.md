# 线程操作

执行下面的代码，创建多个子线程并运行。

```python
from threading import Thread


def func1():
    print('Hello World!')


def func2():
    print('Hello Python!')


def func3():
    print(233, 666)


if __name__ == '__main__':
    t1 = Thread(target=func1)
    t2 = Thread(target=func2)
    t3 = Thread(target=func3)
    t1.start()
    t2.start()
    t3.start()
```

执行下面的代码，利用线程阻塞来确保线程执行与完成的先后顺序。

```python
from threading import Thread


def func1():
    num = 0
    for i in range(1000000):
        num = i
    print('func1', num)


def func2():
    num = 0
    for i in range(1000000):
        num = i
    print('func2', num)


def func3():
    num = 0
    for i in range(1000000):
        num = i
    print('func3', num)


if __name__ == '__main__':
    t1 = Thread(target=func1)
    t2 = Thread(target=func2)
    t3 = Thread(target=func3)
    t1.start()
    t1.join()
    t2.start()
    t2.join()
    t3.start()
```

如果多个线程共同对某个数据进行修改，则可能出现不可预料的结果。

```python
from threading import Thread

num = 0


def func1():
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)


def func2():
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)


def func3():
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)


if __name__ == '__main__':
    t1 = Thread(target=func1)
    t2 = Thread(target=func2)
    t3 = Thread(target=func3)
    t1.start()
    t2.start()
    t3.start()
```

为了保证数据的正确性，需要对多个线程进行同步。

```python
import threading
from threading import Thread

num = 0


def func1():
    lock.acquire()
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)
    lock.release()


def func2():
    lock.acquire()
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)
    lock.release()


def func3():
    lock.acquire()
    global num
    for i in range(1000000):
        num += 1
    print('func1', num)
    lock.release()


if __name__ == '__main__':
    lock = threading.Lock()
    t1 = Thread(target=func1)
    t2 = Thread(target=func2)
    t3 = Thread(target=func3)
    t1.start()
    t2.start()
    t3.start()
```
